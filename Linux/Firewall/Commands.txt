# ==========================================
# LINUX FIREWALL - UFW COMMANDS
# ==========================================

# Check UFW status
sudo ufw status

# Detailed status
sudo ufw status verbose

# Numbered rules
sudo ufw status numbered

# Application profiles
sudo ufw app list

# Allow SSH
sudo ufw allow 22/tcp

# Allow OpenSSH profile
sudo ufw allow OpenSSH

# Allow HTTP
sudo ufw allow 80/tcp

# Allow HTTPS
sudo ufw allow 443/tcp

# Allow custom port
sudo ufw allow 8080/tcp

# Deny a port
sudo ufw deny 3306/tcp

# Delete a rule
sudo ufw delete allow 8080/tcp

# Allow specific IP
sudo ufw allow from 192.168.1.50 to any port 22 proto tcp

# Allow network
sudo ufw allow from 192.168.1.0/24 to any port 22 proto tcp

# Deny specific IP
sudo ufw deny from 192.168.1.100

# Enable firewall
sudo ufw enable

# Disable firewall
sudo ufw disable

# Reset firewall rules
sudo ufw reset

# Enable logging
sudo ufw logging on

# Disable logging
sudo ufw logging off


# ==========================================
# IPTABLES - BASIC INSPECTION
# ==========================================

# List rules
sudo iptables -L

# Detailed rules
sudo iptables -L -n -v

# Show NAT rules
sudo iptables -t nat -L -n -v


# ==========================================
# NETWORK TROUBLESHOOTING
# ==========================================

# Show listening ports
ss -tuln

# Show listening TCP/UDP sockets with processes
sudo ss -tulpn

# Test local HTTP service
curl http://localhost:8080

# Test TCP port
nc -zv localhost 8080
